// 2. إضافة useAuth
import { useAuth } from 'contexts/JWTContext';
import { hasPermission } from 'constants/permissions.constants';

const AuditPage = () => {
  const { user } = useAuth();
  
  const canExport = () => {
    if (!user) return false;
    return hasPermission(user, PERMISSIONS.EXPORT_AUDIT);
  };
  
  // استخدام canExport() في UI
}

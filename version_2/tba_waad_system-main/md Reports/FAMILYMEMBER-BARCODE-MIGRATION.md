ALTER TABLE family_members
ADD COLUMN barcode VARCHAR(100) UNIQUE;

-- Create index for fast barcode lookups
CREATE UNIQUE INDEX idx_family_members_barcode ON family_members(barcode);

-- Generate barcodes for existing family members
-- Format: WAD-{YEAR}-{RANDOM 8 DIGITS}
UPDATE family_members
SET barcode = 'WAD-' || 
              EXTRACT(YEAR FROM CURRENT_DATE) || '-' || 
              LPAD(FLOOR(RANDOM() * 100000000)::TEXT, 8, '0')
WHERE barcode IS NULL;

-- Make barcode NOT NULL after backfilling
ALTER TABLE family_members
ALTER COLUMN barcode SET NOT NULL;

-- Add comment
COMMENT ON COLUMN family_members.barcode IS 
'Auto-generated unique barcode for QR scanning and eligibility verification. Format: WAD-YYYY-NNNNNNNN';

     * Auto-generated by BarcodeGeneratorService

    // Birth date - OPTIONAL

    private LocalDate birthDate;

    @Builder.Default
    @Column(nullable = false, length = 10)
    private Gender gender = Gender.UNDEFINED;

    @PrePersist
    public void ensureDefaults() {
        // CRITICAL: Validate barcode exists
        if (this.barcode == null || this.barcode.trim().isEmpty()) {

        // Set default gender if not provided
        if (this.gender == null) {
            this.gender = Gender.UNDEFINED;
        }

@Service
@RequiredArgsConstructor
public class UnifiedEligibilityService {
    
    private final MemberRepository memberRepository;
    private final FamilyMemberRepository familyMemberRepository; // ADD

    public EligibilityResultDto checkEligibility(String query) {
        EligibilityInputType inputType = detectInputType(query);
        
        switch (inputType) {
            case CARD_NUMBER:
                return checkByCardNumber(query);
            case BARCODE:
                return checkByBarcode(query); // UPDATED to include dependents
            default:
                throw new InvalidEligibilityInputException(
                    "Invalid input format. Use card number (digits only) or barcode (WAD-YYYY-NNNNNNNN)"
                );
        }
    }

    private EligibilityResultDto checkByBarcode(String barcode) {
        // Try primary member first
        Optional<Member> memberOpt = memberRepository.findByBarcode(barcode);
        
        if (memberOpt.isPresent()) {
            return buildEligibilityResult(memberOpt.get(), EligibilityInputType.BARCODE);
        }
        
        // CRITICAL: Try family member (dependent)
        Optional<FamilyMember> dependentOpt = familyMemberRepository.findByBarcode(barcode);
        
        if (dependentOpt.isPresent()) {
            FamilyMember dependent = dependentOpt.get();
            Member primaryMember = dependent.getMember(); // Get primary member for policy info
            
            return buildEligibilityResultForDependent(dependent, primaryMember, EligibilityInputType.BARCODE);
        }
        
        throw new MemberNotFoundException(
            "No member or dependent found with barcode: " + barcode
        );
    }

    /**
     * Build eligibility result for dependent (uses primary member's policy)
     */
    private EligibilityResultDto buildEligibilityResultForDependent(
            FamilyMember dependent,
            Member primaryMember,
            EligibilityInputType inputType) {
        
        // Dependent uses primary member's policy
        BenefitPolicy policy = primaryMember.getBenefitPolicy();
        
        return EligibilityResultDto.builder()
            .memberId(dependent.getId()) // Dependent ID
            .fullName(dependent.getFullName())
            .cardNumber(dependent.getCardNumber())
            .barcode(dependent.getBarcode())
            .memberStatus(dependent.getStatus().name())
            .cardStatus("ACTIVE") // Dependents don't have separate card status
            .eligible(determineEligibility(dependent, primaryMember))
            .employerName(primaryMember.getEmployer() != null ? 
                         primaryMember.getEmployer().getName() : null)
            .policyName(policy != null ? policy.getName() : null)
            .copayAmount(policy != null ? policy.getDefaultCoveragePercent() : null)
            .coverageLimit(policy != null ? policy.getAnnualLimit() : null)
            .message("Dependent of " + primaryMember.getFullName())
            .inputType(inputType)
            .ineligibilityReason(null) // Set if not eligible
            .build();
    }

    private boolean determineEligibility(FamilyMember dependent, Member primaryMember) {
        // Dependent eligibility depends on:
        // 1. Dependent status is ACTIVE
        // 2. Primary member is ACTIVE and ELIGIBLE
        // 3. Policy is ACTIVE
        
        if (dependent.getStatus() != FamilyMemberStatus.ACTIVE) {
            return false;
        }
        
        if (primaryMember.getStatus() != MemberStatus.ACTIVE) {
            return false;
        }
        
        if (!primaryMember.getEligible()) {
            return false;
        }
        
        BenefitPolicy policy = primaryMember.getBenefitPolicy();
        if (policy == null || !"ACTIVE".equals(policy.getStatus())) {
            return false;
        }
        
        return true;
    }
    
    // ... existing methods ...
}
